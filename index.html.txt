<script>
function calculate() {
    let jobAllowance = parseFloat(document.getElementById('jobTitle').value) || 0;
    let baseSalary = parseFloat(document.getElementById('baseSalary').value) || 0;
    let familyAllowance = parseFloat(document.getElementById('familyAllowance').value) || 0;

    let social = baseSalary * 0.07;
    let stamp = baseSalary * 0.008;
    let gross = familyAllowance + jobAllowance + baseSalary;
    
    let taxableBase = (baseSalary + jobAllowance + familyAllowance) - (stamp + social);
    let tax = Math.max(0, (taxableBase - 12560) * 0.15);
    
    let totalDeductions = social + stamp + tax;
    let netSalary = gross - totalDeductions;

    document.getElementById('resJobAllowance').innerText = jobAllowance.toFixed(2);
    document.getElementById('resGross').innerText = gross.toFixed(2);
    document.getElementById('resSocial').innerText = social.toFixed(2);
    document.getElementById('resStamp').innerText = stamp.toFixed(2);
    document.getElementById('resTax').innerText = tax.toFixed(2);
    document.getElementById('resDeductions').innerText = totalDeductions.toFixed(2);
    document.getElementById('resNet').innerText = netSalary.toFixed(2);
}

calculate();
</script>
